/// «Pidele a Tumbao»: una frase en castellano -> filtros sobre la biblioteca.
///
/// Sin red y sin IA: reglas. Se busca la frase mas larga primero y lo que no
/// se entiende se ignora, pero lo entendido se ensenia en chips para que se
/// pueda corregir.
library;

const Map<String, String> _tildes = <String, String>{
  'á': 'a', 'é': 'e', 'í': 'i', 'ó': 'o', 'ú': 'u', 'ü': 'u', 'ñ': 'n',
};

String normalizar(String frase) {
  String t = frase.toLowerCase().split('').map((String c) => _tildes[c] ?? c).join();
  t = t.replaceAll(RegExp(r"\bpa'?l\b"), 'para el').replaceAll(RegExp(r"\bpa'?\s"), 'para ');
  t = t.replaceAll(RegExp(r"[^a-z0-9\s]"), ' ').replaceAll(RegExp(r'\s+'), ' ').trim();
  return ' $t ';
}

enum Orden { tal, dj, subir, bajar, arco, masEscuchadas, recientes, aleatorio }

enum Calidad { alta, sinPerdida, hiRes }

class Peticion {
  double? energiaMin, energiaMax; // percentiles de la biblioteca, 0..1
  double? bpmMin, bpmMax;
  final Set<String> artistas = <String>{};
  final Set<String> sinArtistas = <String>{};
  final Set<String> generos = <String>{};
  final Set<String> sinGeneros = <String>{};
  bool favoritas = false, sinEscuchar = false, olvidadas = false;
  int? minEscuchas;
  Calidad? calidad;
  int? duracionMin, duracionMax; // segundos
  int? dias; // descargada hace como mucho
  int? anioMin, anioMax;
  int? limite, minutos;
  Orden orden = Orden.tal;
  final List<String> entendido = <String>[];

  void energia(double? mn, double? mx) {
    if (mn != null) energiaMin = energiaMin == null ? mn : (mn > energiaMin! ? mn : energiaMin);
    if (mx != null) energiaMax = energiaMax == null ? mx : (mx < energiaMax! ? mx : energiaMax);
  }

  void ordenSiNoHay(Orden o) {
    if (orden == Orden.tal) orden = o;
  }

  @override
  String toString() => entendido.join(' · ');
}

const List<String> negaciones = <String>['sin', 'nada de', 'menos', 'excepto', 'que no sea', 'no', 'ni'];

final Map<RegExp, (String Function(Match), Regla Function(Match))> patrones = <RegExp, (String Function(Match), Regla Function(Match))>{
  RegExp(r' entre (\d{2,3}) y (\d{2,3}) bpm '): (
    (Match m) => '${m[1]}-${m[2]} BPM',
    (Match m) => (Peticion p) { p.bpmMin = double.parse(m[1]!); p.bpmMax = double.parse(m[2]!); },
  ),
  RegExp(r' (?:a )?(\d{2,3}) bpm '): (
    (Match m) => '${m[1]} BPM',
    (Match m) => (Peticion p) { final double b = double.parse(m[1]!); p.bpmMin = b * .97; p.bpmMax = b * 1.03; },
  ),
  RegExp(r' mas de (\d{1,2}) min(?:utos)? '): (
    (Match m) => '+${m[1]} min',
    (Match m) => (Peticion p) => p.duracionMin = int.parse(m[1]!) * 60,
  ),
  RegExp(r' menos de (\d{1,2}) min(?:utos)? '): (
    (Match m) => '-${m[1]} min',
    (Match m) => (Peticion p) => p.duracionMax = int.parse(m[1]!) * 60,
  ),
  RegExp(r' (\d{1,3}) (?:canciones|temas|rolas) '): (
    (Match m) => '${m[1]} canciones',
    (Match m) => (Peticion p) => p.limite = int.parse(m[1]!),
  ),
  RegExp(r' (una|1|2|dos|media) horas? '): (
    (Match m) => m[1] == 'media' ? '30 min' : '${m[1] == 'una' || m[1] == '1' ? 1 : 2} h',
    (Match m) => (Peticion p) => p.minutos = m[1] == 'media' ? 30 : (m[1] == 'una' || m[1] == '1' ? 60 : 120),
  ),
  RegExp(r' (?:de los|anos|los) (?:19)?([5-9]0)s? '): (
    (Match m) => 'Anos ${m[1]}',
    (Match m) => (Peticion p) { p.anioMin = 1900 + int.parse(m[1]!); p.anioMax = p.anioMin! + 9; },
  ),
  RegExp(r' (?:de los|anos) (2000|2010|2020)s? '): (
    (Match m) => 'Anos ${m[1]}',
    (Match m) => (Peticion p) { p.anioMin = int.parse(m[1]!); p.anioMax = p.anioMin! + 9; },
  ),
  RegExp(r' del (19[5-9]\d|20[0-3]\d) '): (
    (Match m) => 'Del ${m[1]}',
    (Match m) => (Peticion p) { p.anioMin = int.parse(m[1]!); p.anioMax = p.anioMin; },
  ),

/// Entiende [frase]. [artistas] son los artistas de la biblioteca, ya normalizados.
Peticion interpretar(String frase, {Set<String> artistas = const <String>{}}) {
  final Peticion p = Peticion();
  String t = normalizar(frase);

  void quitar(int inicio, int fin) => t = '${t.substring(0, inicio)} ${t.substring(fin)}';

  bool negado(int inicio) {
    final String antes = t.substring(0, inicio).trimRight();
    return negaciones.any((String n) => antes.endsWith(' $n') || antes == n);
  }

  void quitarNegacion(int inicio) {
    final String antes = t.substring(0, inicio).trimRight();
    for (final String n in negaciones) {
      if (antes.endsWith(' $n')) {
        t = '${antes.substring(0, antes.length - n.length)} ${t.substring(inicio)}';
        return;
      }
    }
  }

  // 1. Numeros (bpm, minutos, cuantas, anios): antes que nada.
  for (final MapEntry<RegExp, (String Function(Match), Regla Function(Match))> e in patrones.entries) {
    Match? m;
    while ((m = e.key.firstMatch(t)) != null) {
      e.value.$2(m!)(p);
      p.entendido.add(e.value.$1(m));
      quitar(m.start, m.end - 1);
    }
  }
  // 2. Artistas de la biblioteca, el nombre mas largo primero.
  final List<String> porLargo = artistas.where((String a) => a.length >= 3).toList()
    ..sort((String a, String b) => b.length.compareTo(a.length));
  for (final String a in porLargo) {
    final int i = t.indexOf(' $a ');
    if (i < 0) continue;
    if (negado(i)) {
      p.sinArtistas.add(a);
      p.entendido.add('Sin $a');
      quitar(i, i + a.length + 1);
      quitarNegacion(i);
    } else {
      p.artistas.add(a);
      p.entendido.add('De $a');
      quitar(i, i + a.length + 1);
    }
  }
  // 3. Frases fijas, de la mas larga a la mas corta («sin perdida» antes que «sin»).
  final List<String> claves = <String>[...reglas.keys, ...generos.keys]
    ..sort((String a, String b) => b.length.compareTo(a.length));
  for (final String c in claves) {
    final int i = t.indexOf(' $c ');
    if (i < 0) continue;
    if (generos.containsKey(c)) {
      final String g = generos[c]!.$1;
      if (negado(i)) {
        p.sinGeneros.add(g);
        p.entendido.add('Sin $g');
        quitarNegacion(i);
      } else {
        p.generos.add(g);
        p.entendido.add(g[0].toUpperCase() + g.substring(1));
      }
    } else {
      final (String etiqueta, Regla regla) = reglas[c]!;
      regla(p);
      if (!p.entendido.contains(etiqueta)) p.entendido.add(etiqueta);
    }
    final int j = t.indexOf(' $c ');
    if (j >= 0) quitar(j, j + c.length + 1);
  }
  return p;
}
