import sys, subprocess, time
sys.path.insert(0, 'proto')
import ritmo, pulsos
CASOS = {'audio/house_128.flac': (128, 2.0), 'audio/dembow_95.flac': (95, 2.0), 'audio/house_123.5.flac': (123.5, 2.0),
         'audio/trap_140.flac': (140, 2.0), 'audio/balada_72.flac': (72, 2.0), 'audio/largo_house_124.mp3': (124, 0.0)}
for r, (bpm_real, intro) in CASOS.items():
    pcm = subprocess.run(ritmo.orden_ffmpeg('ffmpeg', r), capture_output=True).stdout
    e = ritmo.energias_por_cuadro(pcm)
    bruto = ritmo.flujo(e)
    o = ritmo.quitar_tendencia(bruto)
    bpm, conf = ritmo.estimar_tempo(o)
    P = 6000 / bpm
    n = len(o)
    t0 = time.perf_counter()
    fin = pulsos.pulsos_dp(o, P, n - 3000, n)
    ini = pulsos.pulsos_dp(o, P, 0, 3000)
    t1 = time.perf_counter()
    b = 60 / bpm_real
    def err(fr):
        s = fr / 100 - intro
        k = round(s / b)
        return (s - k * b) * 1000
    def resumen(lista):
        es = [err(f) for f in lista if f / 100 >= intro]
        return f"{len(es)} pulsos, error medio {sum(abs(x) for x in es)/max(1,len(es)):.0f} ms, max {max((abs(x) for x in es), default=0):.0f} ms"
    print(f"{r[6:]:22s} bpm={bpm:.1f} inicio: {resumen(ini)} | fin: {resumen(fin)} | dp {1000*(t1-t0):.0f} ms")
