 'audio/house_128.flac': 128, 'audio/house_123.5.flac': 123.5, 'audio/dembow_95.flac': 95, 'audio/dembow_100.flac': 100,
 'audio/balada_72.flac': 72, 'audio/dnb_174.flac': 174, 'audio/trap_140.flac': 140, 'audio/largo_house_124.mp3': 124,
 'real/admiralbob77_-_Choice_-_Drum-bass.ogg': 136, 'real/Kevin_MacLeod_-_P_I_Tchaikovsky_Dance_of_the_Sugar_Plum_Fairy.ogg': 107.7,

cache = 'onsets.pkl'
if os.path.exists(cache):
    datos = pickle.load(open(cache, 'rb'))
else:
    datos = {}
    for r in VERDAD:
        pcm = subprocess.run(ritmo.orden_ffmpeg('ffmpeg', r), capture_output=True).stdout
        datos[r] = ritmo.energias_por_cuadro(pcm)
    pickle.dump(datos, open(cache, 'wb'))

def interp(a, l):
    i = int(l)
    if i + 1 >= len(a): return 0.0
    f = l - i
    return a[i]*(1-f) + a[i+1]*f

def tempo(e, comb, sigma, pesos, centro=120.0):
    o = ritmo.quitar_tendencia(ritmo.flujo(e, pesos))
    acf = ritmo.autocorrelacion(o, 410)
    mejor = None
    b = 60.0
    while b <= 200:
        lag = 6000 / b
        s = sum(w * interp(acf, k * lag) for k, w in comb)
        s *= math.exp(-0.5 * (math.log2(b / centro) / sigma) ** 2)
        if mejor is None or s > mejor[1]: mejor = (b, s)
        b += 0.5
    return mejor[0]

def acierto(est, real):
    a1 = abs(est - real) / real <= 0.04
    a2 = any(abs(est - real * m) / (real * m) <= 0.04 for m in (0.5, 1, 2, 1/3, 3))
    return a1, a2

VARIANTES = {
 'k1': [(1, 1.0)],
 'k12': [(1, 1.0), (2, 0.5)],
 'k124': [(1, 1.0), (2, 0.5), (4, 0.25)],
 'actual': [(1, 1.0), (2, 0.5), (3, 0.25), (4, 0.25)],
 'k1234eq': [(1, 1.0), (2, 1.0), (3, 1.0), (4, 1.0)],
 'k12_half': [(0.5, 0.5), (1, 1.0), (2, 0.5)],
}
for nombre, comb in VARIANTES.items():
    for sigma in (0.7, 1.0, 1.4):
        for pesos in ((1, 1, 0.6), (1, 1, 1), (1.5, 1, 0.5)):
            a1 = a2 = 0
            fila = []
            for r, real in VERDAD.items():
                est = tempo(datos[r], comb, sigma, pesos)
                x1, x2 = acierto(est, real)
                a1 += x1; a2 += x2
                fila.append(f"{est:.0f}{'' if x1 else '*'}")
            print(f"{nombre:9s} s={sigma} p={pesos} acc1={a1}/{len(VERDAD)} acc2={a2}/{len(VERDAD)}  {' '.join(fila)}")
